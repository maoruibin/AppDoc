if (typeof document !== 'undefined') (function() {
  var KEY = 'gh-density';
  function getSaved() {
    try { return localStorage.getItem(KEY) || 'cozy'; } catch (err) { return 'cozy'; }
  }
  // 事件委托: 首页 DOM 随 SPA 路由重建, 委托到 document 只需绑一次
  document.addEventListener('click', function(e) {
    var b = e.target && e.target.closest && e.target.closest('[data-gh-d]');
    if (!b) return;
    var d = b.getAttribute('data-gh-d');
    try { localStorage.setItem(KEY, d); } catch (err) {}
    applyDensity(d);
  });
  function applyDensity(d) {
    var sections = document.getElementById('gh-sections');
    if (!sections) return;
    sections.classList.toggle('gh-compact', d === 'compact');
    var seg = document.getElementById('gh-density-seg');
    if (seg) {
      for (var i = 0; i < seg.children.length; i++) {
        var b = seg.children[i];
        b.classList.toggle('gh-on', b.getAttribute('data-gh-d') === d);
      }
    }
  }
  // 回到首页时恢复已保存的视图
  setInterval(function() {
    var sections = document.getElementById('gh-sections');
    if (sections && !sections.__ghApplied) {
      sections.__ghApplied = true;
      applyDensity(getSaved());
    }
  }, 500);
  if (document.readyState !== 'loading') {
    var s = document.getElementById('gh-sections');
    if (s) { s.__ghApplied = true; applyDensity(getSaved()); }
  }
})();
</script>
